class S extends ChangeNotifier {
  late SharedPreferences p;
  List<M> notes = [], rems = [], tasks = [];
  String theme = 'system';
  int early = 15;
  bool paused = false;
  bool _b = false, _a = false;
  List<M> _l(String k) => (jsonDecode(p.getString(k) ?? '[]') as List).map((e) => Map<String, dynamic>.from(e)).toList();
  Future<void> load() async {
    p = await SharedPreferences.getInstance();
    notes = _l('notes'); rems = _l('rems'); tasks = _l('tasks');
    theme = p.getString('theme') ?? 'system'; early = p.getInt('early') ?? 15; paused = p.getBool('paused') ?? false;
    resched();
  }
  void save({bool sched = true}) {
    p.setString('notes', jsonEncode(notes)); p.setString('rems', jsonEncode(rems)); p.setString('tasks', jsonEncode(tasks));
    p.setString('theme', theme); p.setInt('early', early); p.setBool('paused', paused);
    notifyListeners();
    if (sched) resched();
  }
  List<M> todayTasks() => tasks.where((t) => (t['days'] as List).contains(DateTime.now().weekday)).toList()
    ..sort((a, b) => mins(a).compareTo(mins(b)));
  int streak() {
    int n = 0;
    for (var i = 0; i < 365; i++) {
      final d = DateTime.now().subtract(Duration(days: i));
      final ts = tasks.where((t) => (t['days'] as List).contains(d.weekday)).toList();
      if (ts.isEmpty) continue;
      if (ts.every((t) => (t['done'] as List).contains(key(d)))) { n++; } else if (i > 0) { break; }
    }
    return n;
  }
  Future<void> askPerms() async {
    final a = plugin.resolvePlatformSpecificImplementation<AndroidFlutterLocalNotificationsPlugin>();
    await a?.requestNotificationsPermission();
    await a?.requestExactAlarmsPermission();
  }
  Future<String> status() async {
    final a = plugin.resolvePlatformSpecificImplementation<AndroidFlutterLocalNotificationsPlugin>();
    final n = await a?.areNotificationsEnabled(), e = await a?.canScheduleExactNotifications();
    return 'Notifications: ${n == true ? 'allowed' : 'blocked'}\nExact alarms: ${e == true ? 'allowed' : 'not allowed'}';
  }
  Future<void> resched() async {
    if (_b) { _a = true; return; }
    _b = true;
    do { _a = false; await _run(); } while (_a);
    _b = false;
  }
  Future<void> _z(int id, String t, String b, tz.TZDateTime at, DateTimeComponents? mc, bool u, AndroidScheduleMode m) =>
      plugin.zonedSchedule(id, t, b, at, nd(u), androidScheduleMode: m,
          uiLocalNotificationDateInterpretation: UILocalNotificationDateInterpretation.absoluteTime, matchDateTimeComponents: mc);
  Future<void> _run() async {
    try {
      await plugin.cancelAll();
      int n = 1;
      final now = tz.TZDateTime.now(tz.local);
      Future<void> go(String t, String b, tz.TZDateTime at, DateTimeComponents? mc, bool u) async {
        if (mc == null && !at.isAfter(now)) return;
        final id = n++;
        try { await _z(id, t, b, at, mc, u, AndroidScheduleMode.exactAllowWhileIdle); }
        catch (_) { try { await _z(id, t, b, at, mc, u, AndroidScheduleMode.inexactAllowWhileIdle); } catch (_) {} }
      }
      final mcs = [null, DateTimeComponents.time, DateTimeComponents.dayOfWeekAndTime, DateTimeComponents.dayOfMonthAndTime, DateTimeComponents.dateAndTime];
      for (final r in rems) {
        if (r['done'] == true) continue;
        final at = tz.TZDateTime.from(DateTime.parse(r['when']), tz.local);
        final mc = mcs[r['repeat'] as int];
        await go(r['title'], '${r['desc']}', at, mc, (r['pri'] as int) >= 2);
        final e = r['early'] as int;
        if (e > 0) await go('In $e min: ${r['title']}', '${r['desc']}', at.subtract(Duration(minutes: e)), mc, false);
      }
      if (!paused) {
        for (final t in tasks) {
          for (final d in (t['days'] as List)) {
            var x = tz.TZDateTime(tz.local, now.year, now.month, now.day, t['h'], t['m']);
            while (x.weekday != d || x.isBefore(now)) { x = tz.TZDateTime(tz.local, x.year, x.month, x.day + 1, t['h'], t['m']); }
            await go(t['title'], 'Routine task', x, DateTimeComponents.dayOfWeekAndTime, t['alarm'] == true);
          }
        }
      }
    } catch (_) {}
  }
}

String cd(DateTime t) {
  final d = t.difference(DateTime.now()), a = d.abs();
  final x = a.inDays > 0 ? '${a.inDays}d ${a.inHours % 24}h' : a.inHours > 0 ? '${a.inHours}h ${a.inMinutes % 60}m' : '${a.inMinutes}m';
  return d.isNegative ? '$x overdue' : 'in $x';
}
DateTime nextOcc(M r) {
  var d = DateTime.parse(r['when']); final n = DateTime.now(); int g = 0;
  while (r['repeat'] != 0 && d.isBefore(n) && g++ < 4000) {
    d = [d, DateTime(d.year, d.month, d.day + 1, d.hour, d.minute), DateTime(d.year, d.month, d.day + 7, d.hour, d.minute),
      DateTime(d.year, d.month + 1, d.day, d.hour, d.minute), DateTime(d.year + 1, d.month, d.day, d.hour, d.minute)][r['repeat'] as int];
  }
  return d;
}

  Widget build(BuildContext context) => L((c) {
    final all = s.rems.toList()..sort((a, b) => nextOcc(a).compareTo(nextOcc(b)));
    final act = all.where((r) => r['done'] != true).toList(), dn = all.where((r) => r['done'] == true).toList();
    return Scaffold(backgroundColor: Colors.transparent,
      floatingActionButton: FloatingActionButton.extended(onPressed: () => editRem(c), icon: const Icon(Icons.add_alert), label: const Text('Reminder')),
      body: ListView(padding: const EdgeInsets.fromLTRB(16, 16, 16, 90), children: [
        hd('Reminders & alarms'), if (act.isEmpty) empty('No reminders. Tap + to add one.'), for (final r in act) remTile(c, r),
        if (dn.isNotEmpty) hd('Completed'), for (final r in dn) remTile(c, r)]));
  });
}
